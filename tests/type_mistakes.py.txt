"""Mistakes the API must reject, run through ty by tests/test_negative_typing.py.

Every line marked `# mistake:` must be a ty error and no other line may be one.
Lines marked `# known miss:` are mistakes ty does not catch today (design Risks);
they must stay silent, so a ty upgrade that starts catching one is visible.
"""

from dataclasses import dataclass

from robotic import Frame, Rotation, Scalar, Transform, UnitVector, Vector


class Base(Frame):
    """Fixed frame."""


class Link1(Frame):
    """Frame of the first link."""


class Tool(Frame):
    """Frame of the tool."""


class Prime(Frame):
    """Intermediate frame."""


class Second(Frame):
    """Second intermediate frame."""


@dataclass(frozen=True)
class Parameters:
    """Joint values."""

    q1: float


q1 = Scalar.symbol("q1")
tip = Tool.point(1, 0, 0)
approach = Tool.vector(1, 0, 0)
base_link1 = Rotation.about_z(q1, target=Base, source=Link1)
base_link1_pose = Transform(target=Base, source=Link1).with_rotation(base_link1)
base_prime = Rotation.about_z(q1, target=Base, source=Prime)
second_tool = Rotation.about_x(q1, target=Second, source=Tool)
compiled_tip = tip.lambdify(Parameters)
compiled_tip_result = compiled_tip(Parameters(q1=0.5))

q1 + "a"  # mistake: not a number
tip + tip  # mistake: point plus point
2 * tip  # mistake: scaling a point
-tip  # mistake: negating a point
Tool.vector(1, 0, 0) + Base.vector(1, 0, 0)  # mistake: mixing frames
tip - Base.origin()  # mistake: point minus point of another frame
approach_in_base: Vector[Base] = approach  # mistake: widening a bound value
approach_anywhere: Vector[Frame] = approach  # mistake: widening a bound value to Frame
widened: Vector[Frame] = Tool.vector(1, 0, 0)  # known miss: explicit inline widening
UnitVector(1, 0, 0)  # mistake: unit vectors come only from a vector
class Misspelled(Frame, notaton="1"): ...  # mistake: misspelled notation keyword
class WrongName(Frame, notation=1): ...  # mistake: a notation must be a string
base_link1 @ base_link1  # mistake: chain Base<-Link1 then Base<-Link1
base_link1 @ Base.vector(1, 0, 0)  # mistake: rotation expects a Link1 vector
Rotation.about_z(q1, target=Base, source=Link1) @ Base.vector(1, 0, 0)  # mistake: inline wrong-frame vector
base_link1 @ Link1.point(1, 0, 0)  # mistake: a rotation does not act on points
base_prime @ second_tool  # mistake: skipped intermediate frame
Rotation.about_axis(Base.vector(0, 0, 1).normalize(), q1, target=Base)  # mistake: the axis fixes the reference frame
base_link1.entry(0, 2)  # mistake: entries are counted from 1
base_link1.entry(4, 1)  # mistake: a rotation has three rows
base_link1.entry(1, int("3"))  # mistake: an int index is not a valid entry index
Transform(target=Base, source=Link1).with_rotation(Rotation.about_z(q1, target=Base, source=Tool))  # mistake: rotation in the wrong frames, inline
Transform(target=Base, source=Link1).with_translation(Link1.vector(0, 0, 1))  # mistake: translation in the wrong frame, inline
base_link1_pose.with_translation(Link1.vector(0, 0, 1))  # mistake: translation in the wrong frame, bound
base_link1_pose.with_translation(Base.origin())  # mistake: a translation is a vector, not a point
base_link1.with_origin(Base.origin())  # mistake: a rotation has no origin; build a Transform
Transform(base_link1, 0, 0, 1)  # mistake: a transform is built from its frames
base_link1_pose @ Tool.point(1, 0, 0)  # mistake: transform expects a Link1 point
base_link1_pose @ Base.origin()  # mistake: transform expects a Link1 point, not a Base one
base_link1_pose @ base_link1_pose  # mistake: chain Base<-Link1 then Base<-Link1
base_link1_pose.entry(5, 1)  # mistake: a transform has four rows
compiled_tip(0.5)  # mistake: values must be a Parameters
compiled_tip(Parameters())  # mistake: missing q1
compiled_tip_result.valid()  # mistake: only axis-angle solutions carry a flag
q1.lambdify(Parameters)(Parameters(q1=0.5)).valid()  # mistake: a scalar carries no flag
