"""Mistakes the API must reject, run by tests/test_negative_typing.py through ty and Python.

Markers, one per line:
- `# mistake:` a ty error, and the line raises when executed (design D35);
- `# static only:` a ty error that cannot raise, because Python does not check
  annotations; the mistake surfaces at the first misuse;
- `# runtime only:` silent for ty, raises when executed (a stored frame differs from
  the annotated one);
- `# known miss:` a mistake ty does not catch today (design Risks); it must stay
  silent, so a ty upgrade that starts catching one is visible.
Every other line must be ty-clean and run without raising.
"""

import math
from dataclasses import dataclass

from robotic import Direction, Displacement, Frame, Position, Rotation, Scalar, Symbol, Transform


class Base(Frame):
    """Fixed frame."""


class Link1(Frame):
    """Frame of the first link."""


class Tool(Frame):
    """Frame of the tool."""


class Prime(Frame):
    """Intermediate frame."""


class Second(Frame):
    """Second intermediate frame."""


class B1(Frame):
    """First frame of the user's probe."""


class B2(Frame):
    """Second frame of the user's probe."""


@dataclass(frozen=True)
class Parameters:
    """Joint values."""

    q1: float


q1 = Symbol("q1")
tip = Tool.position(1, 0, 0)
approach = Tool.displacement(1, 0, 0)
base_link1 = Rotation.about_z(q1, target=Base, source=Link1)
base_link1_pose = base_link1.homogeneous(Base.position(0, 0, 1))
base_prime = Rotation.about_z(q1, target=Base, source=Prime)
second_tool = Rotation.about_x(q1, target=Second, source=Tool)
compiled_tip = tip.compile(Parameters)
compiled_tip_result = compiled_tip(Parameters(q1=0.5))


def needs_a_variable(variable: Symbol) -> Symbol:
    """A function that needs a variable, as a derivative will."""
    return variable


q1 + "a"  # mistake: not a number
q1 + 0.1  # mistake: a float is not exact
Rotation.about_z(math.pi, target=Base, source=Link1)  # mistake: math.pi is not pi
tip + tip  # mistake: position plus position
2 * tip  # mistake: scaling a position
-tip  # mistake: negating a position
approach * approach  # mistake: a displacement times a displacement
Tool.displacement(1, 0, 0) + Base.displacement(1, 0, 0)  # mistake: mixing frames
tip - Base.origin()  # mistake: position minus position of another frame
approach_in_base: Displacement[Scalar, Base] = approach  # static only: widening a bound value
approach_anywhere: Displacement[Scalar, Frame] = approach  # static only: widening a bound value to Frame
widened: Displacement[Scalar, Frame] = Tool.displacement(1, 0, 0)  # known miss: explicit inline widening
widened + Base.displacement(1, 0, 0)  # mistake: the widened value is in Tool
v1: Displacement[Scalar, B2] = Displacement(0, 0, 1, B1)  # static only: annotated in a frame it is not in
v2 = Displacement(0, 0, 1, B2)
v1 + v2  # runtime only: v1 holds B1, whatever its annotation
v2 + v1  # runtime only: v1 holds B1, whatever its annotation
assert repr(v1 + v1) == "Displacement[B1](0, 0, 2)"
Direction(1, 0, 0)  # mistake: a direction comes only from a displacement

base_link1 @ Link1.position(1, 0, 0)  # mistake: a rotation does not act on positions
base_prime @ second_tool  # mistake: skipped intermediate frame
Rotation.about_axis(Base.direction(0, 0, 1), q1, target=Base)  # mistake: the axis fixes the reference frame
base_link1.to_axis_angle(free_axis=Link1.direction(1, 0, 0))  # mistake: the free axis is resolved in Base
base_link1.to_axis_angle(free_axis=Base.displacement(1, 0, 0))  # mistake: the free axis is a direction
base_link1.entry(0, 2)  # mistake: entries are counted from 1
base_link1.entry(4, 1)  # mistake: a rotation has three rows
base_link1.entry(1, int("3"))  # static only: an int index is not a valid entry index

Transform(base_link1, 0, 0, 1)  # mistake: a transform is the homogeneous matrix of a rotation
Transform(base_link1)  # mistake: a transform is built only by homogeneous
